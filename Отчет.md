class Tomato:
    # Статические стадии созревания помидора
    ripening_stages = {0: 'отсутствует', 1: 'цветение', 2: 'зеленый', 3: 'красный'}

    def __init__(self, tomato_index):
        self.tomato_index = tomato_index  # Индекс помидора
        self.current_stage = 0  # Текущая стадия созревания

    def grow(self):
        if self.current_stage < 3:
            self.current_stage += 1

    def is_ripe(self):

        return self.current_stage == 3

class TomatoBush:
    def __init__(self, number_of_tomatoes):
        self.tomatoes_list = [Tomato(tomato_index) for tomato_index in range(number_of_tomatoes)]

    def grow_all_tomatoes(self):
        """ Переводит все томаты на кусте на следующую стадию созревания. """
        for tomato in self.tomatoes_list:
            tomato.grow()

    def are_all_tomatoes_ripe(self):
        return all(tomato.is_ripe() for tomato in self.tomatoes_list)

    def harvest_all_tomatoes(self):
        """ Сбор урожая: очистка списка томатов после сбора. """
        self.tomatoes_list = []

class Gardener:
    def __init__(self, gardener_name, tomato_bush):
        self.gardener_name = gardener_name
        self.tomato_bush = tomato_bush

    def take_care_of_plants(self):
        """ Садовник ухаживает за растением, переводя все томаты на следующую стадию созревания. """
        self.tomato_bush.grow_all_tomatoes()

    def collect_harvest(self):
        """ Проверка созревания всех томатов и сбор урожая. """
        if self.tomato_bush.are_all_tomatoes_ripe():
            self.tomato_bush.harvest_all_tomatoes()
            print("Урожай собран!")
        else:
            print("Предупреждение: не все томаты созрели!")

    @staticmethod
    def gardening_guide():
        """ Вывод справки по садоводству. """
        print("Справка по садоводству: Садовод должен ухаживать за растением, чтобы оно выросло.")

# Вызов справки по садоводству
Gardener.gardening_guide()

# Создание куста с 4 помидорами
tomato_bush = TomatoBush(4)

# Создание садовника, который будет ухаживать за кустом
gardener = Gardener("Иван", tomato_bush)

# Уход за кустом с помидорами
gardener.take_care_of_plants()
gardener.take_care_of_plants()
gardener.take_care_of_plants()

# Попытка собрать урожай, когда томаты еще не созрели
gardener.collect_harvest()

# Продолжение ухода за кустом
gardener.take_care_of_plants()

# Сбор урожая
